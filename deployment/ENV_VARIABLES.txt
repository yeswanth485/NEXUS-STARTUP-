# QUICK REFERENCE - ENVIRONMENT VARIABLES TO SET
# ================================================
# Copy these to your hosting platforms

# ── VERCEL (Frontend) ──────────────────────────
# Go to: https://vercel.com/dashboard > Your Project > Settings > Environment Variables

NEXT_PUBLIC_SUPABASE_URL=https://YOUR_PROJECT_ID.supabase.co
NEXT_PUBLIC_SUPABASE_ANON_KEY=YOUR_ANON_KEY
NEXT_PUBLIC_API_URL=https://YOUR_BACKEND_NAME.onrender.com

# ── RENDER (Backend) ───────────────────────────
# Go to: https://dashboard.render.com > Your Service > Environment

PORT=5000
CLIENT_URL=https://YOUR_FRONTEND_NAME.vercel.app
SUPABASE_URL=https://YOUR_PROJECT_ID.supabase.co
SUPABASE_ANON_KEY=YOUR_ANON_KEY
SUPABASE_SERVICE_ROLE_KEY=YOUR_SERVICE_ROLE_KEY
RAZORPAY_KEY_ID=rzp_test_xxxxxxxxxxxx
RAZORPAY_KEY_SECRET=YOUR_RAZORPAY_SECRET

# ── WHERE TO FIND EACH VALUE ───────────────────
#
# Supabase:
#   Dashboard > Settings > API
#   - Project URL    = NEXT_PUBLIC_SUPABASE_URL = SUPABASE_URL
#   - anon (public)  = NEXT_PUBLIC_SUPABASE_ANON_KEY = SUPABASE_ANON_KEY
#   - service_role   = SUPABASE_SERVICE_ROLE_KEY (SECRET - never expose to frontend!)
#
# Razorpay:
#   Dashboard > Settings > API Keys
#   - Key ID     = RAZORPAY_KEY_ID
#   - Key Secret = RAZORPAY_KEY_SECRET
#
# Backend URL:
#   After Render deploys: https://YOUR_SERVICE_NAME.onrender.com
#   Set this as NEXT_PUBLIC_API_URL on Vercel AND as CLIENT_URL on Render
#
# Frontend URL:
#   After Vercel deploys: https://YOUR_PROJECT_NAME.vercel.app
#   Set this as CLIENT_URL on Render
